# Likely cause
The cluster has no pull credentials for your
private registry.

# Fix
kubectl create secret docker-registry regcred \
  --docker-server=ghcr.io \
  --docker-username=$GH_USER \
  --docker-password=$GH_TOKEN

# Then reference it in the pod spec
imagePullSecrets:
  - name: regcred
